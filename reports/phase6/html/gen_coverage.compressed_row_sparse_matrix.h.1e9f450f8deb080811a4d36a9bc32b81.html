165 // Options struct to control the generation of random block sparse
166 // matrices in compressed row sparse format.
167 //
168 // The random matrix generation proceeds as follows.
169 //
170 // First the row and column block structure is determined by
171 // generating random row and column block sizes that lie within the
172 // given bounds.
173 //
174 // Then we walk the block structure of the resulting matrix, and with
175 // probability block_density determine whether they are structurally
176 // zero or not. If the answer is no, then we generate entries for the
177 // block which are distributed normally.
178 struct RandomMatrixOptions {
179 // Type of matrix to create.
180 //
181 // If storage_type is UPPER_TRIANGULAR (LOWER_TRIANGULAR), then
182 // create a square symmetric matrix with just the upper triangular
183 // (lower triangular) part. In this case, num_col_blocks,
184 // min_col_block_size and max_col_block_size will be ignored and
185 // assumed to be equal to the corresponding row settings.
186 StorageType storage_type = StorageType::UNSYMMETRIC;
187
188 int num_row_blocks = 0;
189 int min_row_block_size = 0;
190 int max_row_block_size = 0;
191 int num_col_blocks = 0;
192 int min_col_block_size = 0;
193 int max_col_block_size = 0;
194
195 // 0 < block_density <= 1 is the probability of a block being
196 // present in the matrix. A given random matrix will not have
197 // precisely this density.
198 double block_density = 0.0;
199 };
