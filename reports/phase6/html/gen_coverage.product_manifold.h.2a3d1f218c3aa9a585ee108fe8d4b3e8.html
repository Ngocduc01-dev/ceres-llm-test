52 // Construct a manifold by taking the Cartesian product of a number of other
53 // manifolds. This is useful, when a parameter block is the Cartesian product
54 // of two or more manifolds. For example the parameters of a camera consist of
55 // a rotation and a translation, i.e., SO(3) x R^3.
56 //
57 // Example usage:
58 //
59 // ProductManifold<QuaternionManifold, EuclideanManifold<3>> se3;
60 //
61 // is the manifold for a rigid transformation, where the rotation is
62 // represented using a quaternion.
63 //
64 // Manifolds can be copied and moved to ProductManifold:
65 //
66 // SubsetManifold manifold1(5, {2});
67 // SubsetManifold manifold2(3, {0, 1});
68 // ProductManifold<SubsetManifold, SubsetManifold> manifold(manifold1,
69 // manifold2);
70 //
71 // In advanced use cases, manifolds can be dynamically allocated and passed as
72 // (smart) pointers:
73 //
74 // ProductManifold<std::unique_ptr<QuaternionManifold>, EuclideanManifold<3>>
75 // se3{std::make_unique<QuaternionManifold>(), EuclideanManifold<3>{}};
76 //
77 // In C++17, the template parameters can be left out as they are automatically
78 // deduced making the initialization much simpler:
79 //
80 // ProductManifold se3{QuaternionManifold{}, EuclideanManifold<3>{}};
81 //
82 // The manifold implementations must be either default constructible, copyable
83 // or moveable to be usable in a ProductManifold.
84 template <typename Manifold0, typename Manifold1, typename... ManifoldN>
85 class ProductManifold final : public Manifold {

257 template <typename T, std::size_t N>
258 static std::array<T, N> ExclusiveScan(const std::array<T, N>& values) {
259 std::array<T, N> result;
260 // TODO Replace with std::exclusive_scan once all platforms have full C++17
261 // STL support.
262 T init = 0;
263 for (std::size_t i = 0; i != N; ++i) {
264 result[i] = init;
265 init += values[i];
266 }
267 return result;
268 }
