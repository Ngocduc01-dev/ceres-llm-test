40 // The (L)BFGS algorithm explicitly requires that the secant equation:
41 //
42 // B_{k+1} * s_k = y_k
43 //
44 // Is satisfied at each iteration, where B_{k+1} is the approximated
45 // Hessian at the k+1-th iteration, s_k = (x_{k+1} - x_{k}) and
46 // y_k = (grad_{k+1} - grad_{k}). As the approximated Hessian must be
47 // positive definite, this is equivalent to the condition:
48 //
49 // s_k^T * y_k > 0 [s_k^T * B_{k+1} * s_k = s_k^T * y_k > 0]
50 //
51 // This condition would always be satisfied if the function was strictly
52 // convex, alternatively, it is always satisfied provided that a Wolfe line
53 // search is used (even if the function is not strictly convex). See [1]
54 // (p138) for a proof.
55 //
56 // Although Ceres will always use a Wolfe line search when using (L)BFGS,
57 // practical implementation considerations mean that the line search
58 // may return a point that satisfies only the Armijo condition, and thus
59 // could violate the Secant equation. As such, we will only use a step
60 // to update the Hessian approximation if:
61 //
62 // s_k^T * y_k > tolerance
63 //
64 // It is important that tolerance is very small (and >=0), as otherwise we
65 // might skip the update too often and fail to capture important curvature
66 // information in the Hessian. For example going from 1e-10 -> 1e-14 improves
67 // the NIST benchmark score from 43/54 to 53/54.
68 //
69 // [1] Nocedal J., Wright S., Numerical Optimization, 2nd Ed. Springer, 1999.
70 //
71 // TODO(alexs.mac): Consider using Damped BFGS update instead of
72 // skipping update.
73 const double kLBFGSSecantConditionHessianUpdateTolerance = 1e-14;

87 bool LowRankInverseHessian::Update(const Vector& delta_x,
88 const Vector& delta_gradient) {
89 const double delta_x_dot_delta_gradient = delta_x.dot(delta_gradient);
90 if (delta_x_dot_delta_gradient <=
91 kLBFGSSecantConditionHessianUpdateTolerance) {
92 VLOG(2) << "Skipping L-BFGS Update, delta_x_dot_delta_gradient too "
93 << "small: " << delta_x_dot_delta_gradient
94 << ", tolerance: " << kLBFGSSecantConditionHessianUpdateTolerance
95 << " (Secant condition).";
96 return false;
97 }
98
99 int next = indices_.size();
100 // Once the size of the list reaches max_num_corrections_, simulate
101 // a circular buffer by removing the first element of the list and
102 // making it the next position where the LBFGS history is stored.
103 if (next == max_num_corrections_) {
104 next = indices_.front();
105 indices_.pop_front();
106 }
107
108 indices_.push_back(next);
109 delta_x_history_.col(next) = delta_x;
110 delta_gradient_history_.col(next) = delta_gradient;
111 delta_x_dot_delta_gradient_(next) = delta_x_dot_delta_gradient;
112 approximate_eigenvalue_scale_ =
113 delta_x_dot_delta_gradient / delta_gradient.squaredNorm();
114 return true;
115 }

117 void LowRankInverseHessian::RightMultiplyAndAccumulate(const double* x_ptr,
118 double* y_ptr) const {
119 ConstVectorRef gradient(x_ptr, num_parameters_);
120 VectorRef search_direction(y_ptr, num_parameters_);
121
122 search_direction = gradient;
123
124 const int num_corrections = indices_.size();
125 Vector alpha(num_corrections);
126
127 for (auto it = indices_.rbegin(); it != indices_.rend(); ++it) {
128 const double alpha_i = delta_x_history_.col(*it).dot(search_direction) /
129 delta_x_dot_delta_gradient_(*it);
130 search_direction -= alpha_i * delta_gradient_history_.col(*it);
131 alpha(*it) = alpha_i;
132 }
133
134 if (use_approximate_eigenvalue_scaling_) {
135 // Rescale the initial inverse Hessian approximation (H_0) to be iteratively
136 // updated so that it is of similar 'size' to the true inverse Hessian along
137 // the most recent search direction. As shown in [1]:
138 //
139 // \gamma_k = (delta_gradient_{k-1}' * delta_x_{k-1}) /
140 // (delta_gradient_{k-1}' * delta_gradient_{k-1})
141 //
142 // Satisfies:
143 //
144 // (1 / \lambda_m) <= \gamma_k <= (1 / \lambda_1)
145 //
146 // Where \lambda_1 & \lambda_m are the smallest and largest eigenvalues of
147 // the true Hessian (not the inverse) along the most recent search direction
148 // respectively. Thus \gamma is an approximate eigenvalue of the true
149 // inverse Hessian, and choosing: H_0 = I * \gamma will yield a starting
150 // point that has a similar scale to the true inverse Hessian. This
151 // technique is widely reported to often improve convergence, however this
152 // is not universally true, particularly if there are errors in the initial
153 // jacobians, or if there are significant differences in the sensitivity
154 // of the problem to the parameters (i.e. the range of the magnitudes of
155 // the components of the gradient is large).
156 //
157 // The original origin of this rescaling trick is somewhat unclear, the
158 // earliest reference appears to be Oren [1], however it is widely discussed
159 // without specific attribution in various texts including [2] (p143/178).
160 //
161 // [1] Oren S.S., Self-scaling variable metric (SSVM) algorithms Part II:
162 // Implementation and experiments, Management Science,
163 // 20(5), 863-874, 1974.
164 // [2] Nocedal J., Wright S., Numerical Optimization, Springer, 1999.
165 search_direction *= approximate_eigenvalue_scale_;
166
167 VLOG(4) << "Applying approximate_eigenvalue_scale: "
168 << approximate_eigenvalue_scale_ << " to initial inverse Hessian "
169 << "approximation.";
170 }
171
172 for (const int i : indices_) {
173 const double beta = delta_gradient_history_.col(i).dot(search_direction) /
174 delta_x_dot_delta_gradient_(i);
175 search_direction += delta_x_history_.col(i) * (alpha(i) - beta);
176 }
177 }
