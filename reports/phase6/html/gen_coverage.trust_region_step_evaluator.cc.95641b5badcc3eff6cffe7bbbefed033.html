69 void TrustRegionStepEvaluator::StepAccepted(const double cost,
70 const double model_cost_change) {
71 // Algorithm 10.1.2 from Trust Region Methods by Conn, Gould &
72 // Toint.
73 //
74 // Step 3a
75 current_cost_ = cost;
76 accumulated_candidate_model_cost_change_ += model_cost_change;
77 accumulated_reference_model_cost_change_ += model_cost_change;
78
79 // Step 3b.
80 if (current_cost_ < minimum_cost_) {
81 minimum_cost_ = current_cost_;
82 num_consecutive_nonmonotonic_steps_ = 0;
83 candidate_cost_ = current_cost_;
84 accumulated_candidate_model_cost_change_ = 0.0;
85 } else {
86 // Step 3c.
87 ++num_consecutive_nonmonotonic_steps_;
88 if (current_cost_ > candidate_cost_) {
89 candidate_cost_ = current_cost_;
90 accumulated_candidate_model_cost_change_ = 0.0;
91 }
92 }
93
94 // Step 3d.
95 //
96 // At this point we have made too many non-monotonic steps and
97 // we are going to reset the value of the reference iterate so
98 // as to force the algorithm to descend.
99 //
100 // Note: In the original algorithm by Toint, this step was only
101 // executed if the step was non-monotonic, but that would not handle
102 // the case of max_consecutive_nonmonotonic_steps = 0. The small
103 // modification of doing this always handles that corner case
104 // correctly.
105 if (num_consecutive_nonmonotonic_steps_ ==
106 max_consecutive_nonmonotonic_steps_) {
107 reference_cost_ = candidate_cost_;
108 accumulated_reference_model_cost_change_ =
109 accumulated_candidate_model_cost_change_;
110 }
111 }
