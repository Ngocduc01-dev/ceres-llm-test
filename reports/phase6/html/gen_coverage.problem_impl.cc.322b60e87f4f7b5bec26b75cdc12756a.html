68 // Returns true if two regions of memory, a and b, with sizes size_a and size_b
69 // respectively, overlap.
70 bool RegionsAlias(const double* a, int size_a, const double* b, int size_b) {
71 return (a < b) ? b < (a + size_a) : a < (b + size_b);
72 }

91 if (it->second == 1) {

122 ParameterBlock* ProblemImpl::InternalAddParameterBlock(double* values,
123 int size) {
124 CHECK(values != nullptr) << "Null pointer passed to AddParameterBlock "
125 << "for a parameter with size " << size;
126
127 // Ignore the request if there is a block for the given pointer already.
128 auto it = parameter_block_map_.find(values);
129 if (it != parameter_block_map_.end()) {
130 if (!options_.disable_all_safety_checks) {
131 int existing_size = it->second->Size();
132 CHECK(size == existing_size)
133 << "Tried adding a parameter block with the same double pointer, "
134 << values << ", twice, but with different block sizes. Original "
135 << "size was " << existing_size << " but new size is " << size;
136 }
137 return it->second;
138 }
139
140 if (!options_.disable_all_safety_checks) {
141 // Before adding the parameter block, also check that it doesn't alias any
142 // other parameter blocks.
143 if (!parameter_block_map_.empty()) {
144 auto lb = parameter_block_map_.lower_bound(values);
145
146 // If lb is not the first block, check the previous block for aliasing.
147 if (lb != parameter_block_map_.begin()) {
148 auto previous = lb;
149 --previous;
150 CheckForNoAliasing(
151 previous->first, previous->second->Size(), values, size);
152 }
153
154 // If lb is not off the end, check lb for aliasing.
155 if (lb != parameter_block_map_.end()) {
156 CheckForNoAliasing(lb->first, lb->second->Size(), values, size);
157 }
158 }
159 }
160
161 // Pass the index of the new parameter block as well to keep the index in
162 // sync with the position of the parameter in the program's parameter vector.
163 auto* new_parameter_block =
164 new ParameterBlock(values, size, program_->parameter_blocks_.size());
165
166 // For dynamic problems, add the list of dependent residual blocks, which is
167 // empty to start.
168 if (options_.enable_fast_removal) {
169 new_parameter_block->EnableResidualBlockDependencies();
170 }
171 parameter_block_map_[values] = new_parameter_block;
172 program_->parameter_blocks_.push_back(new_parameter_block);
173 return new_parameter_block;
174 }

176 void ProblemImpl::InternalRemoveResidualBlock(ResidualBlock* residual_block) {
177 CHECK(residual_block != nullptr);
178 // Perform no check on the validity of residual_block, that is handled in
179 // the public method: RemoveResidualBlock().
180
181 // If needed, remove the parameter dependencies on this residual block.
182 if (options_.enable_fast_removal) {
183 const int num_parameter_blocks_for_residual =
184 residual_block->NumParameterBlocks();
185 for (int i = 0; i < num_parameter_blocks_for_residual; ++i) {
186 residual_block->parameter_blocks()[i]->RemoveResidualBlock(
187 residual_block);
188 }
189
190 auto it = residual_block_set_.find(residual_block);
191 residual_block_set_.erase(it);
192 }
193 DeleteBlockInVector(program_->mutable_residual_blocks(), residual_block);
194 }
195
196 // Deletes the residual block in question, assuming there are no other
197 // references to it inside the problem (e.g. by another parameter). Referenced
198 // cost and loss functions are tucked away for future deletion, since it is not
199 // possible to know whether other parts of the problem depend on them without
200 // doing a full scan.
201 void ProblemImpl::DeleteBlock(ResidualBlock* residual_block) {
202 // The const casts here are legit, since ResidualBlock holds these
203 // pointers as const pointers but we have ownership of them and
204 // have the right to destroy them when the destructor is called.
205 auto* cost_function =
206 const_cast<CostFunction*>(residual_block->cost_function());
207 if (options_.cost_function_ownership == TAKE_OWNERSHIP) {
208 DecrementValueOrDeleteKey(cost_function, &cost_function_ref_count_);
209 }
210
211 auto* loss_function =
212 const_cast<LossFunction*>(residual_block->loss_function());
213 if (options_.loss_function_ownership == TAKE_OWNERSHIP &&
214 loss_function != nullptr) {
215 DecrementValueOrDeleteKey(loss_function, &loss_function_ref_count_);
216 }
217
218 delete residual_block;
219 }

221 // Deletes the parameter block in question, assuming there are no other
222 // references to it inside the problem (e.g. by any residual blocks).
223 void ProblemImpl::DeleteBlock(ParameterBlock* parameter_block) {
224 parameter_block_map_.erase(parameter_block->mutable_user_state());
225 delete parameter_block;
226 }

267 ResidualBlockId ProblemImpl::AddResidualBlock(
268 CostFunction* cost_function,
269 LossFunction* loss_function,
270 double* const* const parameter_blocks,
271 int num_parameter_blocks) {
272 CHECK(cost_function != nullptr);
273 CHECK_EQ(num_parameter_blocks, cost_function->parameter_block_sizes().size());
274
275 // Check the sizes match.
276 const std::vector<int32_t>& parameter_block_sizes =
277 cost_function->parameter_block_sizes();
278
279 if (!options_.disable_all_safety_checks) {
280 CHECK_EQ(parameter_block_sizes.size(), num_parameter_blocks)
281 << "Number of blocks input is different than the number of blocks "
282 << "that the cost function expects.";
283
284 // Check for duplicate parameter blocks.
285 std::vector<double*> sorted_parameter_blocks(
286 parameter_blocks, parameter_blocks + num_parameter_blocks);
287 sort(sorted_parameter_blocks.begin(), sorted_parameter_blocks.end());
288 const bool has_duplicate_items =
289 (std::adjacent_find(sorted_parameter_blocks.begin(),
290 sorted_parameter_blocks.end()) !=
291 sorted_parameter_blocks.end());
292 if (has_duplicate_items) {
293 std::string blocks;
294 for (int i = 0; i < num_parameter_blocks; ++i) {
295 blocks += StringPrintf(" %p ", parameter_blocks[i]);
296 }
297
298 LOG(FATAL) << "Duplicate parameter blocks in a residual parameter "
299 << "are not allowed. Parameter block pointers: [" << blocks
300 << "]";
301 }
302 }
303
304 // Add parameter blocks and convert the double*'s to parameter blocks.
305 std::vector<ParameterBlock*> parameter_block_ptrs(num_parameter_blocks);
306 for (int i = 0; i < num_parameter_blocks; ++i) {
307 parameter_block_ptrs[i] = InternalAddParameterBlock(
308 parameter_blocks[i], parameter_block_sizes[i]);
309 }
310
311 if (!options_.disable_all_safety_checks) {
312 // Check that the block sizes match the block sizes expected by the
313 // cost_function.
314 for (int i = 0; i < parameter_block_ptrs.size(); ++i) {
315 CHECK_EQ(cost_function->parameter_block_sizes()[i],
316 parameter_block_ptrs[i]->Size())
317 << "The cost function expects parameter block " << i << " of size "
318 << cost_function->parameter_block_sizes()[i]
319 << " but was given a block of size "
320 << parameter_block_ptrs[i]->Size();
321 }
322 }
323
324 auto* new_residual_block =
325 new ResidualBlock(cost_function,
326 loss_function,
327 parameter_block_ptrs,
328 program_->residual_blocks_.size());
329
330 // Add dependencies on the residual to the parameter blocks.
331 if (options_.enable_fast_removal) {
332 for (int i = 0; i < num_parameter_blocks; ++i) {
333 parameter_block_ptrs[i]->AddResidualBlock(new_residual_block);
334 }
335 }
336
337 program_->residual_blocks_.push_back(new_residual_block);
338
339 if (options_.enable_fast_removal) {
340 residual_block_set_.insert(new_residual_block);
341 }
342
343 if (options_.cost_function_ownership == TAKE_OWNERSHIP) {
344 // Increment the reference count, creating an entry in the table if
345 // needed. Note: C++ maps guarantee that new entries have default
346 // constructed values; this implies integers are zero initialized.
347 ++cost_function_ref_count_[cost_function];
348 }
349
350 if (options_.loss_function_ownership == TAKE_OWNERSHIP &&
351 loss_function != nullptr) {
352 ++loss_function_ref_count_[loss_function];
353 }
354
355 return new_residual_block;
356 }

378 // Delete a block from a vector of blocks, maintaining the indexing invariant.
379 // This is done in constant time by moving an element from the end of the
380 // vector over the element to remove, then popping the last element. It
381 // destroys the ordering in the interest of speed.
382 template <typename Block>
383 void ProblemImpl::DeleteBlockInVector(std::vector<Block*>* mutable_blocks,
384 Block* block_to_remove) {
385 CHECK_EQ((*mutable_blocks)[block_to_remove->index()], block_to_remove)
386 << "You found a Ceres bug! \n"
387 << "Block requested: " << block_to_remove->ToString() << "\n"
388 << "Block present: "
389 << (*mutable_blocks)[block_to_remove->index()]->ToString();
390
391 // Prepare the to-be-moved block for the new, lower-in-index position by
392 // setting the index to the blocks final location.
393 Block* tmp = mutable_blocks->back();
394 tmp->set_index(block_to_remove->index());
395
396 // Overwrite the to-be-deleted residual block with the one at the end.
397 (*mutable_blocks)[block_to_remove->index()] = tmp;
398
399 DeleteBlock(block_to_remove);
400
401 // The block is gone so shrink the vector of blocks accordingly.
402 mutable_blocks->pop_back();
403 }

405 void ProblemImpl::RemoveResidualBlock(ResidualBlock* residual_block) {
406 CHECK(residual_block != nullptr);
407
408 // Verify that residual_block identifies a residual in the current problem.
409 const std::string residual_not_found_message = StringPrintf(
410 "Residual block to remove: %p not found. This usually means "
411 "one of three things have happened:\n"
412 " 1) residual_block is uninitialised and points to a random "
413 "area in memory.\n"
414 " 2) residual_block represented a residual that was added to"
415 " the problem, but referred to a parameter block which has "
416 "since been removed, which removes all residuals which "
417 "depend on that parameter block, and was thus removed.\n"
418 " 3) residual_block referred to a residual that has already "
419 "been removed from the problem (by the user).",
420 residual_block);
421 if (options_.enable_fast_removal) {
422 CHECK(residual_block_set_.find(residual_block) != residual_block_set_.end())
423 << residual_not_found_message;
424 } else {
425 // Perform a full search over all current residuals.
426 CHECK(std::find(program_->residual_blocks().begin(),
427 program_->residual_blocks().end(),
428 residual_block) != program_->residual_blocks().end())
429 << residual_not_found_message;
430 }
431
432 InternalRemoveResidualBlock(residual_block);
433 }

435 void ProblemImpl::RemoveParameterBlock(const double* values) {
436 ParameterBlock* parameter_block = FindWithDefault(
437 parameter_block_map_, const_cast<double*>(values), nullptr);
438 if (parameter_block == nullptr) {
439 LOG(FATAL) << "Parameter block not found: " << values
440 << ". You must add the parameter block to the problem before "
441 << "it can be removed.";
442 }
443
444 if (options_.enable_fast_removal) {
445 // Copy the dependent residuals from the parameter block because the set of
446 // dependents will change after each call to RemoveResidualBlock().
447 std::vector<ResidualBlock*> residual_blocks_to_remove(
448 parameter_block->mutable_residual_blocks()->begin(),
449 parameter_block->mutable_residual_blocks()->end());
450 for (auto* residual_block : residual_blocks_to_remove) {
451 InternalRemoveResidualBlock(residual_block);
452 }
453 } else {
454 // Scan all the residual blocks to remove ones that depend on the parameter
455 // block. Do the scan backwards since the vector changes while iterating.
456 const int num_residual_blocks = NumResidualBlocks();
457 for (int i = num_residual_blocks - 1; i >= 0; --i) {
458 ResidualBlock* residual_block =
459 (*(program_->mutable_residual_blocks()))[i];
460 const int num_parameter_blocks = residual_block->NumParameterBlocks();
461 for (int j = 0; j < num_parameter_blocks; ++j) {
462 if (residual_block->parameter_blocks()[j] == parameter_block) {
463 InternalRemoveResidualBlock(residual_block);
464 // The parameter blocks are guaranteed unique.
465 break;
466 }
467 }
468 }
469 }
470 DeleteBlockInVector(program_->mutable_parameter_blocks(), parameter_block);
471 }

585 bool ProblemImpl::Evaluate(const Problem::EvaluateOptions& evaluate_options,
586 double* cost,
587 std::vector<double>* residuals,
588 std::vector<double>* gradient,
589 CRSMatrix* jacobian) {
590 if (cost == nullptr && residuals == nullptr && gradient == nullptr &&
591 jacobian == nullptr) {
592 return true;
593 }
594
595 // If the user supplied residual blocks, then use them, otherwise
596 // take the residual blocks from the underlying program.
597 Program program;
598 *program.mutable_residual_blocks() =
599 ((evaluate_options.residual_blocks.size() > 0)
600 ? evaluate_options.residual_blocks
601 : program_->residual_blocks());
602
603 const std::vector<double*>& parameter_block_ptrs =
604 evaluate_options.parameter_blocks;
605
606 std::vector<ParameterBlock*> variable_parameter_blocks;
607 std::vector<ParameterBlock*>& parameter_blocks =
608 *program.mutable_parameter_blocks();
609
610 if (parameter_block_ptrs.size() == 0) {
611 // The user did not provide any parameter blocks, so default to
612 // using all the parameter blocks in the order that they are in
613 // the underlying program object.
614 parameter_blocks = program_->parameter_blocks();
615 } else {
616 // The user supplied a vector of parameter blocks. Using this list
617 // requires a number of steps.
618
619 // 1. Convert double* into ParameterBlock*
620 parameter_blocks.resize(parameter_block_ptrs.size());
621 for (int i = 0; i < parameter_block_ptrs.size(); ++i) {
622 parameter_blocks[i] = FindWithDefault(
623 parameter_block_map_, parameter_block_ptrs[i], nullptr);
624 if (parameter_blocks[i] == nullptr) {
625 LOG(FATAL) << "No known parameter block for "
626 << "Problem::Evaluate::Options.parameter_blocks[" << i << "]"
627 << " = " << parameter_block_ptrs[i];
628 }
629 }
630
631 // 2. The user may have only supplied a subset of parameter
632 // blocks, so identify the ones that are not supplied by the user
633 // and are NOT constant. These parameter blocks are stored in
634 // variable_parameter_blocks.
635 //
636 // To ensure that the parameter blocks are not included in the
637 // columns of the jacobian, we need to make sure that they are
638 // constant during evaluation and then make them variable again
639 // after we are done.
640 std::vector<ParameterBlock*> all_parameter_blocks(
641 program_->parameter_blocks());
642 std::vector<ParameterBlock*> included_parameter_blocks(
643 program.parameter_blocks());
644
645 std::vector<ParameterBlock*> excluded_parameter_blocks;
646 sort(all_parameter_blocks.begin(), all_parameter_blocks.end());
647 sort(included_parameter_blocks.begin(), included_parameter_blocks.end());
648 set_difference(all_parameter_blocks.begin(),
649 all_parameter_blocks.end(),
650 included_parameter_blocks.begin(),
651 included_parameter_blocks.end(),
652 back_inserter(excluded_parameter_blocks));
653
654 variable_parameter_blocks.reserve(excluded_parameter_blocks.size());
655 for (auto* parameter_block : excluded_parameter_blocks) {
656 if (!parameter_block->IsConstant()) {
657 variable_parameter_blocks.push_back(parameter_block);
658 parameter_block->SetConstant();
659 }
660 }
661 }
662
663 // Setup the Parameter indices and offsets before an evaluator can
664 // be constructed and used.
665 program.SetParameterOffsetsAndIndex();
666
667 Evaluator::Options evaluator_options;
668
669 // Even though using SPARSE_NORMAL_CHOLESKY requires SuiteSparse or
670 // CXSparse, here it just being used for telling the evaluator to
671 // use a SparseRowCompressedMatrix for the jacobian. This is because
672 // the Evaluator decides the storage for the Jacobian based on the
673 // type of linear solver being used.
674 evaluator_options.linear_solver_type = SPARSE_NORMAL_CHOLESKY;
675 evaluator_options.num_threads = evaluate_options.num_threads;
676
677 // The main thread also does work so we only need to launch num_threads - 1.
678 context_impl_->EnsureMinimumThreads(evaluator_options.num_threads - 1);
679 evaluator_options.context = context_impl_;
680 evaluator_options.evaluation_callback =
681 program_->mutable_evaluation_callback();
682 std::unique_ptr<Evaluator> evaluator(
683 new ProgramEvaluator<ScratchEvaluatePreparer,
684 CompressedRowJacobianWriter>(evaluator_options,
685 &program));
686
687 if (residuals != nullptr) {
688 residuals->resize(evaluator->NumResiduals());
689 }
690
691 if (gradient != nullptr) {
692 gradient->resize(evaluator->NumEffectiveParameters());
693 }
694
695 std::unique_ptr<CompressedRowSparseMatrix> tmp_jacobian;
696 if (jacobian != nullptr) {
697 tmp_jacobian.reset(down_cast<CompressedRowSparseMatrix*>(
698 evaluator->CreateJacobian().release()));
699 }
700
701 // Point the state pointers to the user state pointers. This is
702 // needed so that we can extract a parameter vector which is then
703 // passed to Evaluator::Evaluate.
704 program.SetParameterBlockStatePtrsToUserStatePtrs();
705
706 // Copy the value of the parameter blocks into a vector, since the
707 // Evaluate::Evaluate method needs its input as such. The previous
708 // call to SetParameterBlockStatePtrsToUserStatePtrs ensures that
709 // these values are the ones corresponding to the actual state of
710 // the parameter blocks, rather than the temporary state pointer
711 // used for evaluation.
712 Vector parameters(program.NumParameters());
713 program.ParameterBlocksToStateVector(parameters.data());
714
715 double tmp_cost = 0;
716
717 Evaluator::EvaluateOptions evaluator_evaluate_options;
718 evaluator_evaluate_options.apply_loss_function =
719 evaluate_options.apply_loss_function;
720 bool status =
721 evaluator->Evaluate(evaluator_evaluate_options,
722 parameters.data(),
723 &tmp_cost,
724 residuals != nullptr ? &(*residuals)[0] : nullptr,
725 gradient != nullptr ? &(*gradient)[0] : nullptr,
726 tmp_jacobian.get());
727
728 // Make the parameter blocks that were temporarily marked constant,
729 // variable again.
730 for (auto* parameter_block : variable_parameter_blocks) {
731 parameter_block->SetVarying();
732 }
733
734 if (status) {
735 if (cost != nullptr) {
736 *cost = tmp_cost;
737 }
738 if (jacobian != nullptr) {
739 tmp_jacobian->ToCRSMatrix(jacobian);
740 }
741 }
742
743 program_->SetParameterBlockStatePtrsToUserStatePtrs();
744 program_->SetParameterOffsetsAndIndex();
745 return status;
746 }

748 bool ProblemImpl::EvaluateResidualBlock(ResidualBlock* residual_block,
749 bool apply_loss_function,
750 bool new_point,
751 double* cost,
752 double* residuals,
753 double** jacobians) const {
754 auto evaluation_callback = program_->mutable_evaluation_callback();
755 if (evaluation_callback) {
756 evaluation_callback->PrepareForEvaluation(jacobians != nullptr, new_point);
757 }
758
759 ParameterBlock* const* parameter_blocks = residual_block->parameter_blocks();
760 const int num_parameter_blocks = residual_block->NumParameterBlocks();
761 for (int i = 0; i < num_parameter_blocks; ++i) {
762 ParameterBlock* parameter_block = parameter_blocks[i];
763 if (parameter_block->IsConstant()) {
764 if (jacobians != nullptr && jacobians[i] != nullptr) {
765 LOG(ERROR) << "Jacobian requested for parameter block : " << i
766 << ". But the parameter block is marked constant.";
767 return false;
768 }
769 } else {
770 CHECK(parameter_block->SetState(parameter_block->user_state()))
771 << "Congratulations, you found a Ceres bug! Please report this error "
772 << "to the developers.";
773 }
774 }
775
776 double dummy_cost = 0.0;
777 FixedArray<double, 32> scratch(
778 residual_block->NumScratchDoublesForEvaluate());
779 return residual_block->Evaluate(apply_loss_function,
780 cost ? cost : &dummy_cost,
781 residuals,
782 jacobians,
783 scratch.data());
784 }

865 void ProblemImpl::GetResidualBlocksForParameterBlock(
866 const double* values, std::vector<ResidualBlockId>* residual_blocks) const {
867 ParameterBlock* parameter_block = FindWithDefault(
868 parameter_block_map_, const_cast<double*>(values), nullptr);
869 if (parameter_block == nullptr) {
870 LOG(FATAL) << "Parameter block not found: " << values
871 << ". You must add the parameter block to the problem before "
872 << "you can get the residual blocks that depend on it.";
873 }
874
875 if (options_.enable_fast_removal) {
876 // In this case the residual blocks that depend on the parameter block are
877 // stored in the parameter block already, so just copy them out.
878 CHECK(residual_blocks != nullptr);
879 residual_blocks->resize(parameter_block->mutable_residual_blocks()->size());
880 std::copy(parameter_block->mutable_residual_blocks()->begin(),
881 parameter_block->mutable_residual_blocks()->end(),
882 residual_blocks->begin());
883 return;
884 }
885
886 // Find residual blocks that depend on the parameter block.
887 CHECK(residual_blocks != nullptr);
888 residual_blocks->clear();
889 const int num_residual_blocks = NumResidualBlocks();
890 for (int i = 0; i < num_residual_blocks; ++i) {
891 ResidualBlock* residual_block = (*(program_->mutable_residual_blocks()))[i];
892 const int num_parameter_blocks = residual_block->NumParameterBlocks();
893 for (int j = 0; j < num_parameter_blocks; ++j) {
894 if (residual_block->parameter_blocks()[j] == parameter_block) {
895 residual_blocks->push_back(residual_block);
896 // The parameter blocks are guaranteed unique.
897 break;
898 }
899 }
900 }
